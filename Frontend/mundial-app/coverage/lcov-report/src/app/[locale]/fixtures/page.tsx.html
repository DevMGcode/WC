                  className="relative flex items-center gap-2 px-4 py-2 rounded-xl flex-shrink-0 overflow-hidden"
                  style={{ background: isActive ? `linear-gradient(135deg, ${color}18, rgba(2,8,20,0.90))` : 'transparent', border: `1px solid ${isActive ? color + '35' : 'transparent'}`, cursor: 'pointer', outline: 'none' }}
                  whileTap={{ scale: 0.96 }}>
                  <span style={{ color: isActive ? color : 'rgba(100,116,139,0.60)', fontSize: 13, filter: isActive ? `drop-shadow(0 0 5px ${glow})` : 'none', display: 'flex' }}>{icon}</span>
                  <span className="text-[10px] font-black tracking-[0.1em] whitespace-nowrap"
                    style={{ color: isActive ? color : 'rgba(100,116,139,0.60)', textShadow: isActive ? `0 0 10px ${glow}` : 'none' }}>{t(labelKey)}</span>
                  {count > 0 && (
                    <span className="text-[8px] font-black px-1.5 py-0.5 rounded-full tabular-nums"
                      style={{ background: isActive ? `${color}22` : 'rgba(255,255,255,0.04)', color: isActive ? color : 'rgba(100,116,139,0.45)', border: `1px solid ${isActive ? color + '28' : 'transparent'}` }}>
                      {count}
                    </span>
                  )}
                </motion.button>
              );
            })}
            <div className="flex-1" />
            <div className="hidden sm:flex items-center pr-2">
              <EQBars color="rgba(34,211,238,0.35)" count={6} maxH={14} />
            </div>
          </div>
        </motion.div>
 
        {loading ? (
          <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
            {Array.from({ length: 6 }).map((_, i) => (
              <motion.div key={i} className="rounded-2xl overflow-hidden"
                style={{ background: 'rgba(4,12,28,0.95)', border: '1px solid rgba(34,211,238,0.07)', height: 160 }}
                animate={{ opacity: [0.3, 0.6, 0.3] }} transition={{ duration: 1.8, repeat: Infinity, delay: i * 0.15 }}>
                <div className="h-full flex flex-col">
                  <div className="h-10 border-b" style={{ borderColor: 'rgba(255,255,255,0.04)', background: 'rgba(0,0,0,0.20)' }} />
                  <div className="flex-1 flex items-center justify-around px-6 py-4 gap-4">
                    <div className="w-12 h-12 rounded-full" style={{ background: 'rgba(255,255,255,0.04)' }} />
                    <div className="w-12 h-6 rounded-lg" style={{ background: 'rgba(255,255,255,0.04)' }} />
                    <div className="w-12 h-12 rounded-full" style={{ background: 'rgba(255,255,255,0.04)' }} />
                  </div>
                  <div className="h-8 border-t" style={{ borderColor: 'rgba(255,255,255,0.04)', background: 'rgba(0,0,0,0.20)' }} />
                </div>
              </motion.div>
            ))}
          </div>
        ) : grouped.length > 0 ? (
          <div className="space-y-6">
            {grouped.map(({ key, label, items }) => (
              <motion.div key={key} initial={{ opacity: 0 }} animate={{ opacity: 1 }} transition={{ duration: 0.4 }}>
                <div className="flex items-center gap-3 mb-3">
                  <div className="flex items-center gap-2">
                    <div className="w-1.5 h-1.5 rounded-full bg-cyan-400/60" />
                    <span className="text-[10px] font-black tracking-[0.28em] uppercase text-slate-500">{label}</span>
                  </div>
                  <div className="flex-1 h-px" style={{ background: 'linear-gradient(90deg, rgba(34,211,238,0.12), transparent)' }} />
                  <span className="text-[8px] font-bold text-slate-700 tracking-widest uppercase shrink-0">
                    {items.length} {items.length === 1 ? 'partido' : 'partidos'}
                  </span>
                </div>
                <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
                  <AnimatePresence>
                    {items.map((fixture) => {
                      const idx = globalIdx++;
                      return <MatchCard key={fixture.id} fixture={fixture} index={idx} t={t} />;
                    })}
                  </AnimatePresence>
                </div>
              </motion.div>
            ))}
          </div>
        ) : (
          <motion.div initial={{ opacity: 0, y: 16 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.5 }}
            className="flex flex-col items-center justify-center py-24 gap-5">
            <div className="relative">
              <div className="absolute inset-0 rounded-full pointer-events-none"
                style={{ background: 'rgba(34,211,238,0.10)', filter: 'blur(20px)', transform: 'scale(1.4)' }} />
              <div className="relative w-16 h-16 rounded-2xl flex items-center justify-center"
                style={{ background: 'linear-gradient(145deg, rgba(34,211,238,0.12), rgba(4,12,28,0.90))', border: '1px solid rgba(34,211,238,0.22)' }}>
                <FiCalendar size={28} style={{ color: 'rgba(34,211,238,0.55)' }} />
              </div>
            </div>
            <div className="text-center">
              <p className="text-sm font-black text-slate-400 mb-1">{t('fixtures.noMatches')}</p>
              <p className="text-[11px] text-slate-700">{t('fixtures.noMatches')}</p>
            </div>
            <motion.button onClick={() => setFilter('ALL')}
              className="px-5 py-2 rounded-xl text-[10px] font-black tracking-[0.18em] uppercase"
              style={{ color: '#22d3ee', border: '1px solid rgba(34,211,238,0.22)', background: 'rgba(34,211,238,0.07)' }}
              whileHover={{ scale: 1.05 }} whileTap={{ scale: 0.96 }}>
              Ver todos los partidos
            </motion.button>
          </motion.div>
        )}
      </div>
    </div>
  );
